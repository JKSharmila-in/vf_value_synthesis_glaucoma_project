// ─── Load patient data & fundus image dynamically ─────────────────────────── function loadData() { // 1. Attempt to fetch unified object, or look for standalone individual keys let age = sessionStorage.getItem('age'); let gender = sessionStorage.getItem('gender'); let iop = sessionStorage.getItem('iop'); let cct = sessionStorage.getItem('cct'); // If stored as a single 'patientData' JSON string object instead: const profileString = sessionStorage.getItem('patientData'); if (profileString) { try { const parsed = JSON.parse(profileString); if (parsed.age) age = parsed.age; if (parsed.gender) gender = parsed.gender; if (parsed.iop) iop = parsed.iop; if (parsed.cct) cct = parsed.cct; } catch(e) { console.error("Error parsing patientData object:", e); } } // Fallback check via URL parameters if sessionStorage is clear const urlParams = new URLSearchParams(window.location.search); if (!age && urlParams.has('age')) age = urlParams.get('age'); if (!gender && urlParams.has('gender')) gender = urlParams.get('gender'); if (!iop && urlParams.has('iop')) iop = urlParams.get('iop'); if (!cct && urlParams.has('cct')) cct = urlParams.get('cct'); // 2. Bind the variables directly to UI text nodes (Removing the hardcoded defaults) document.getElementById('dispAge').textContent = age || 'Not Provided'; if (gender) { document.getElementById('dispGender').textContent = (gender.toLowerCase().startsWith('f')) ? 'Female' : 'Male'; } else { document.getElementById('dispGender').textContent = 'Not Provided'; } document.getElementById('dispIOP').textContent = iop ? `${iop}` : '—'; document.getElementById('dispCCT').textContent = cct ? `${cct}` : '—'; // 3. Render Fundus Image if present const fundusData = sessionStorage.getItem('fundusImage'); if (fundusData && fundusData.trim() !== "") { const img = document.getElementById('fundusImg'); img.src = fundusData; img.style.display = 'block'; document.getElementById('fundusPlaceholder').style.display = 'none'; } else { // If truly empty, let's keep placeholder visible but generic document.getElementById('fundusImg').style.display = 'none'; document.getElementById('fundusPlaceholder').style.display = 'flex'; } // 4. Render VF Values matrix const stored = sessionStorage.getItem('vfValues'); const vfValues = stored ? JSON.parse(stored) : DEFAULT_VF; drawVF(vfValues); }
